Hosts Discovery

Escanear Red en busca de Hosts

sudo nmap -sn 192.168.1.0/24

Alternativas

# ARP-SCAN
sudo arp-scan -I <interfaz_de_red> --localnet --ignoredups

# NetDiscover
sudo netdiscover -r 192.168.100.0/24

# FPing
fping -a -g 192.168.100.0/24 2>/dev/null

# NBTscan: nombres NetBIOS en redes Windows
nbtscan 192.168.1.0/24

# Masscan: “el más rápido” 
sudo masscan 192.168.0.0/16 -p1-65535 --rate 50000 --wait 0

host_discovery.sh

#!/bin/bash 

function ctrl_c() {
    echo -e "\n\n[!] Exiting...\n"
    tput cnorm
    exit 1
}

tput civis
trap ctrl_c INT

for i in $(seq 1 254); do
    timeout 1 bash -c "ping -c 1 192.168.1.$i" &> /dev/null && echo -e "[+] Host: 192.168.1.$i - Active" &

done

wait

tput cnorm
