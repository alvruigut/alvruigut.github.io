1521 - Oracle Database

1521 (Oracle Database) Base de datos Oracle.

Comandos en Oracle Database

Comandos

# Iniciar sesión con credenciales válidas. 
sqlplus /<contraseña>@<ip_servidor>:1521/ 

# Iniciar sesión como administrador.
sqlplus / as sysdba  

# Enumerar usuarios de la base de datos.
SELECT username FROM dba_users;  

# Mostrar todas las tablas en la base de datos.
SELECT * FROM all_tables; 

# Enumerar todas las vistas disponibles.
SELECT * FROM all_views;  

# Enumerar todos los índices de la base de datos. 
SELECT * FROM all_indexes; 

# Obtener hashes de contraseñas de los usuarios.
SELECT password FROM sys.user$ WHERE name=''

Enumerar con nmap

oracle

# Buscar todos los scripts que tiene nmap para Oracle.
ls -la /usr/share/nmap/scripts/ | grep -e "oracle"

# Utilizar todos los scripts.
nmap -p 1521 <ip_objetivo> --script "oracle-*"

# Enumeración de usuarios. 
nmap -p 1521 --script oracle-enum-users <ip_victima> 

# Ataque de fuerza bruta.
nmap -p 1521 --script oracle-brute <ip_victima>  

# Descubrimiento del SID de la base de datos.
nmap -p 1521 --script oracle-sid-brute <ip_victima> 

# Ver versión del servicio TNS.
nmap -p 1521 --script oracle-tns-version <ip_victima>
