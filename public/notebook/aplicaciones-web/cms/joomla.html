Identificar Joomla

Nmap

# Enumerar versión y extensiones
nmap -sV --script http-joomla-enum <url>

# Enumerar vulnerabilidades
nmap -sV --script "http-joomla* and not http-joomla-brute" <url>
nmap --script=http-vuln* -p 80,443 <url>

# Enumerar usuarios
nmap -sV --script http-joomla-users <url>
