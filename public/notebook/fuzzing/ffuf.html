FFUF

ffuf

# Versión mínima (barrido simple de directorios)
ffuf -u http://target/FUZZ -w /ruta/wordlist.txt

# Con códigos comunes
ffuf -u http://target/FUZZ -w /ruta/wordlist.txt -mc 200,301,302

# Con extensiones 
ffuf -u http://target/FUZZ -w /ruta/wordlist.txt -e .php,.html,.bak

# Con salida guardada
ffuf -u http://target/FUZZ -w /ruta/wordlist.txt -mc 200,301,302 -o resultados.txt
  • -u URL con FUZZ
  • -w wordlist
  • -t threads
  • -mc match codes (ej. 200,301)
  • -fs filter size
  • -fc filter codes
  • -e extensiones
  • -o salida, -of formato (json, csv, html)
  • -recursion y -recursion-depth
  • -H header, -b cookie, -d data, -X method
