impacket-smbserver

# Desde Linux
impacket-smbserver smbFolder $(pwd) -smb2support 
# Desde Windows
copy "C:\Users\alvar\Pictures\Joker.jpg" \\<kali_ip>\smbFolder\

impacket-smbserver con unidades del almacenamiento

# Desde Linux
# Abrimos servidor smb
impacket-smbserver <smbFolder> $(pwd) -smb2support 

impacket-smbserver <smbFolder> $(pwd) -smb2support -username alvaro -password alvaro123 # Con contraseña

# Desde Windows
# Nos conectamos al servidor smb
net use L: \\<ip_linux>\<smbFolder> /user:alvaro alvaro123
# Mandamos el recurso que queramos
copy "C:\Users\alvar\Pictures\Joker.jpg" L:\
# Lo desmontamos
net use L: /delete
