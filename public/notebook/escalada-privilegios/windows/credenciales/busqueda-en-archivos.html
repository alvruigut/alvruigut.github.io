Búsqueda en archivos

Strings

PowerShell

# String: password   Ruta: C:/users
Get-ChildItem "C:/users" -Recurse -File | Select-String "password"
# Ocultar Errores: -Recurse -File
Get-ChildItem "C:/users" -Recurse -File -ErrorAction SilentlyContinue | Select-String "password"

# Strings: password y user  Ruta: C:/users
Get-ChildItem "C:\users" -Recurse -File | Select-String "password|user"

# String: password Ruta: C:/users Fichero:.bak .txt
Get-ChildItem "C:\users" -Recurse -File -Include *.bak,*.txt | Select-String "password"

# String: password   Ruta: C:/users Formato: Tabla
Get-ChildItem "C:/users" -Recurse -File | Select-String "password" | Format-Table Path,LineNumber,Line

# Strings: cpassword, password, pass, pwd, db, db_pwd, db_pass, user, username, db_user...

CMD

# String: password Directorio actual en fichero con extensiones ...

findstr /SIM /C:"password" *.txt *.ini *.cfg *.config *.xml *.git *.ps1 *.yml

# /S: Recursivo
# /I: Key InSensitive
# /M: Nombre del archivo
# /N: Número línea
# /C: String

# Strings: cpassword, password, pass, pwd, db, db_pwd, db_pass, user, username, db_user...

Ficheros

PowerShell

# Ficheros: .bak .txt   Ruta: C:/users
Get-ChildItem "C:\users" -Recurse -File -Include *.bak,*.txt -ErrorAction SilentlyContinue | Select-Object FullName

# Ficheros:.txt Ruta: C:/users Ocultar: C:\Windows\
Get-ChildItem "C:\users" -Recurse -File -Include *.bak,*.txt -ErrorAction SilentlyContinue | Where-Object { $_.FullName -notlike "C:\Windows\*" } | Select-Object FullName

#*.xml,*.ini,*.txt,*.config,*.kdbx,*.bak,.vmdk,.vdhx,.ppk

CMD

# Ficheros: .bak .txt   Ruta: Actual
dir /s *.txt *.bak
# Desde powershell: cmd /c dir /s *.txt *.bak
