AD Enumeración

Comandos base para obtener contexto del dominio, controladores, usuarios, grupos, equipos, shares y permisos relevantes.

PowerShell

net user /domain # Enumerar usuarios del dominio
net group /domain # Enumerar grupos del dominio
net user alvaro alvaro123 /add /domain # Crear nuevo usuario y añadirlo al dominio
net group "Exchange Windows Permissions" alvaro /add # Añadir al usuario alvaro a un grupo

PowerView.ps1

PowerView

Import-Module .\PowerView.ps1

Get-NetDomain # Información básica del dominio
Get-NetDomainController # Listar DCs
Get-NetPolicy # Políticas de contraseñas

Get-NetUser  # Listar todos los usuarios del dominio
Get-NetUser "user"  # Listar un usuario específico
Get-NetUser | select samaccountname, description, pwdlastset  # Listar usuario con filtros (Descripciones de Usuarios)
Get-ADGroupMember "Domain Admins" # Listar Domain Admins

Get-DomainUser -PreauthNotRequired             # Identifica usuarios AS-REP Roastable
Get-NetUser -SPN                               # Identifica usuarios Kerberoastable
Get-NetUser -SPN | select serviceprincipalname # Identifica usuarios Kerberoastable
Get-NetUser -SPN | select samaccountname,serviceprincipalname  # Cuentas SPN del dominio 

Get-NetGroup # Listar todos los grupos del dominio
Get-NetGroup "group name" # Listar un grupo específico
Get-NetComputer # Enumera Computers Objects del dominio

Find-LocalAdminAccess # Enumera las máquinas del dominio donde el usuario actual es ADMIN LOCAL
Find-DomainShare # Busca Shares del dominio

Get-ObjectAcl -Identity <user> # Enumera ACE, listas SID, objetosSID 
Convert-SidToName <sid/objsid> # Convierte SID a nombre común
Get-ObjectAcl -Identity "group-name" | ? {$_.ActiveDirectoryRights -eq "GenericAll"} | select SecurityIdentifier,ActiveDirectoryRights  # Check del permiso "GenericAll"
