Modo listener y cliente

# Modo Listener
nc -nlvp 4444
# Modo Cliente
nc -nv <ip_destino> 4444

# Si escribo como cliente "Hello" lo recibiré en el Listener

nc para Windows target

rlwrap nc -nlvp 444

# rlwrap permite hacer en una cmd CTRL+L, moverte con las flechitas...

Mandar archivos

# Desde kali mandar un archivo a windows
# Kali (Cliente): Creamos fichero
echo "Hello, this was sent over with Netcat" >> test.txt

# Windows (Listener): Nos ponemos en escucha esperando el fichero test.txt
nc.exe -nvlp 1234 > test.txt

# Kali (Cliente): Mandamos el fichero test.txt
nc -nv 10.0.27.35 1234 < test.txt
