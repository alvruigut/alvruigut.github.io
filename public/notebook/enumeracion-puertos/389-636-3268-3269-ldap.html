389, 636, 3268, 3269 - LDAP

Enumeramos el DC sin credenciales

# Enumeracion anónimo 
ldapsearch -x -H ldap://<dc_ip> -b 'DC=<dominio>,DC=<domimio>'
# Enumeración credenciales
ldapsearch -x -H ldap://<dc_ip> -D '<user>@<domain>' -w 'pass' -b 'DC=<domain>,DC=<domain>'

# Enumeracion usuarios
<ldap_command> -s sub "(&(objectclass=user))" | grep "userPrincipalName"
<ldap_command> | grep "Service Accounts"
<ldap_command> | grep "CN"

# Enumeracion usuarios en condiciones
<ldap_command> | grep -i 'samaccountname: <user>' -C 40

# Enumeracion usuarios y cuentas de máquinas
<ldap_command> | grep -i accountname

# Enumeración contraseñas
<ldap_command> | grep -i "pwd" -C1
<ldap_command> | grep -iE "pwd|desc|password" -C 3

# Filtrar por clase de objeto 
<ldap_command> -W "objectclass=account"
nxc ldap <ip> -u '' -p '' --query "(objectClass=*)" "*"
