gobuster

# Escaneo simple
gobuster dir -u http://target -w /ruta/wordlist.txt

# Escaneo con extensiones
gobuster dir -u http://target -w /ruta/wordlist.txt -x php,html,txt

# Modo `dir` (Enumeración de Directorios y Archivos)
gobuster dir -u <url> -w /usr/share/wordlists/dirbuster/directory-list-2.3-medium.txt -t <hilos>

# Modo `vhost` (Descubrimiento de Virtual Hosts)
gobuster vhost -u https://<IP> -w /usr/share/wordlists/SecLists/Discovery/DNS/subdomains-top1million-5000.txt -t 50 --domain <domain> --append-domain
  • dir modo directorios
  • -u URL
  • -w wordlist
  • -k omitir verificación SSL
  • -b ocultar códigos de estado
  • -t threads
  • -x extensiones a probar (separadas por coma)
  • -s status codes a mostrar
  • -e mostrar resultados “expandido” (show full URL)
  • -r recursión
  • -o exportar los resultados a un archivo.
