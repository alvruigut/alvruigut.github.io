NMAP Port Scan

# Un puerto
nmap -p80 192.168.1.1

# Más de un puerto
nmap -p80,443 192.168.1.1

# Un rango
nmap -p1-900 192.168.1.1

# Los 500 puertos más comunes
nmap  --top-ports 500 192.168.1.1

-p- Escanea todo el rango de puertos.

--open Muestra solo los puertos abiertos.

-sS Realiza un TCP SYN Scan sigiloso.

-sT Realiza un TCP Connect Scan con el flujo completo del three-way handshake.

-sU Realiza un UDP Scan.

--min-rate 5000 Aumenta la velocidad del escaneo. Para eJPT puede ser suficiente --min-rate 2000.

-T5 Ajusta la velocidad de escaneo. T4 < T3 < T2 < T1: a menor velocidad, más sigiloso.

-vvv Modo triple verbose para mayor detalle.

-n Omite la resolución DNS.

-Pn Asume que el host está activo, evitando el ping de descubrimiento.

-oG Exporta resultados en formato grepeable.

-oN Exporta resultados en formato normal.

portscan.sh

#!/bin/bash

function ctrl_c() {
    echo -e "\n\n[!] Exiting...\n"
    tput cnorm
    pkill -P $$ >/dev/null 2>&1 || true
    exit 1

}

trap ctrl_c INT

declare -a ports=( $(seq 1 65535) )

function scan_port() {
    local host="$1"
    local port="$2"
    # Abrir socket TCP; redirigir errores a /dev/null
    (exec 3<>/dev/tcp/"$host"/"$port") 2>/dev/null
    if [ $? -eq 0 ]; then
        echo -e "\n[+] Host: $host - Port: $port (OPEN)"
    fi
    # Cerrar descriptor
    exec 3>&-
    exec 3<&-
}

tput civis

# Comprobar argumento
if [ -n "${1-}" ]; then
    target="$1"
    for port in "${ports[@]}"; do
        printf '\rScanning %s:%s' "$target" "$port" >&2 || true

        scan_port "$target" "$port" &
    done


else
    echo -e "\n[!] Uso: $0 <ip-address>\n"
    tput cnorm
    exit 1
fi

wait
printf '\n' >&2 || true
tput cnorm
