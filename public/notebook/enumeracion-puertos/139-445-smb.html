#!/bin/bash

ip_target="192.168.1.10"
wordlist="/ruta/shares.txt"

while read -r share; do
	if smbclient -N "//$ip_target/$share" &>/dev/null; then
		echo "Recurso con null sesion encontrado: $share"
	fi
done < "$wordlist"
