# Versión normal (barrido simple de directorios)
wfuzz -c -L  -t 40 --hc=404 -w /ruta/wordlist.txt -u "http://target/FUZZ/" 

# Extensiones
wfuzz -c --hc=404 -t 200 -w /usr/share/wordlists/dirbuster/directory-list-2.3-medium.txt -z list,html-asp-txt http://target/FUZZ.FUZ2Z/

# Multiple payloads siguiendo la secuencia numérica FUZ2Z, FUZ3Z...
wfuzz -c -w users.txt -w passwords.txt -u "http://target/login?user=FUZZ&pass=FUZ2Z
  • --hc=404 hide codes (ej. --hc 404)
  • --sc=200 show codes
  • --hh=2048 hide header length
  • --sh=2048 show header length
  • -t threads
  • -c color output
  • -w wordlist
  • -u URL (con FUZZ)
  • -L Aplica follow redirect para evitar redirecciones (301,302) y mostrar 200
  • -H "User-Agent: Google Chrome" Cambia la cabecera del user-agent
  • -z Extensiones
